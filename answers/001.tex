\begin{enumerate}[itemsep=0.9em]

\item \textbf{Змейка:}
\[
15 \xrightarrow{\cdot 4} 60 \xrightarrow{+24} 84
\xrightarrow{:3} 28 \xrightarrow{+17} 45 \xrightarrow{-19} 26
\xrightarrow{\cdot 3} 78 \xrightarrow{:2} 39 \xrightarrow{-14} \boxed{25}.
\]

\item а) $168000$; \qquad б) $200$.

\item А--2; \qquad Б--1; \qquad В--4; \qquad Г--3.

\item Получится $500:20=25$ петушков. Выручка равна
$25\cdot30=750$ рублей, стоимость карамели --- $100$ рублей.

\textbf{Ответ: $650$ рублей.}

\item Длина прямоугольной песочницы равна $12$ дм, ширина --- $8$ дм.
Сторона квадратной песочницы равна $10$ дм. Площади равны $96$ дм$^2$
и $100$ дм$^2$.

\textbf{Ответ: $4$ дм$^2$.}

\item Например:
\[
\text{а)}\quad (9\cdot9+9):9=10;
\qquad
\text{б)}\quad 9+9:9+(9-9)=10.
\]

\item Нужно заменить на умножение знак перед числом $9$:
\[
1+2+3+4+5+6+7+8\cdot9=100.
\]

\end{enumerate}
